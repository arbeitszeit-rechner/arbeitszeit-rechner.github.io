{{- /* JSON-LD als ein @graph pro Seite: Organization + WebSite (auf jeder Seite), WebPage (Untertyp je Seite), BreadcrumbList (außer Startseite), Hauptentität: Article (Startseite), DefinedTermSet (Glossar-Hub) oder DefinedTerm (Glossar-Eintrag). Es werden ausschließlich Daten ausgegeben, die auch sichtbar auf der Seite stehen oder in der Konfiguration gepflegt sind. */ -}} {{- if ne .Kind "404" -}} {{- $home := site.Home.Permalink -}} {{- $lang := site.Params.htmlLang -}} {{- $desc := .Params.description | default site.Params.description -}} {{- $orgID := printf "%s#organization" $home -}} {{- $siteID := printf "%s#website" $home -}} {{- $pageID := printf "%s#webpage" .Permalink -}} {{- $author := site.Data.author -}} {{- $org := dict "@type" "Organization" "@id" $orgID "name" site.Title "url" $home -}} {{- $website := dict "@type" "WebSite" "@id" $siteID "url" $home "name" site.Title "inLanguage" $lang "publisher" (dict "@id" $orgID) -}} {{- $graph := slice $org $website -}} {{- /* Seitentyp und Hauptentität bestimmen */ -}} {{- $ptype := .Params.webpage_type | default "WebPage" -}} {{- $main := .Params.main_entity | default "" -}} {{- if and (eq .Kind "section") (eq .Section "glossar") -}} {{- $ptype = "CollectionPage" -}} {{- $main = "DefinedTermSet" -}} {{- else if and (eq .Kind "page") (eq .Section "glossar") -}} {{- $main = "DefinedTerm" -}} {{- end -}} {{- $page := dict "@type" $ptype "@id" $pageID "url" .Permalink "name" .Title "description" $desc "inLanguage" $lang "isPartOf" (dict "@id" $siteID) "publisher" (dict "@id" $orgID) -}} {{- if not .Date.IsZero -}}{{- $page = merge $page (dict "datePublished" (.Date.Format "2006-01-02")) -}}{{- end -}} {{- if not .Lastmod.IsZero -}}{{- $page = merge $page (dict "dateModified" (.Lastmod.Format "2006-01-02")) -}}{{- end -}} {{- /* Autor: Person nur, wenn in data/author.yaml ein Name steht, sonst die Redaktion (Organization) */ -}} {{- $authorRef := dict "@id" $orgID -}} {{- if $author.name -}} {{- $authorID := printf "%s#author" $home -}} {{- $person := dict "@type" "Person" "@id" $authorID "name" $author.name -}} {{- with $author.url }}{{ $person = merge $person (dict "url" .) }}{{ end -}} {{- with $author.jobTitle }}{{ $person = merge $person (dict "jobTitle" .) }}{{ end -}} {{- with $author.sameAs }}{{ $person = merge $person (dict "sameAs" .) }}{{ end -}} {{- $graph = $graph | append $person -}} {{- $authorRef = dict "@id" $authorID -}} {{- end -}} {{- /* Brotkrumen */ -}} {{- $crumbs := partial "helpers/crumbs.html" . -}} {{- if $crumbs -}} {{- $bcID := printf "%s#breadcrumb" .Permalink -}} {{- $items := slice -}} {{- range $i, $c := $crumbs -}} {{- $items = $items | append (dict "@type" "ListItem" "position" (add $i 1) "name" $c.name "item" $c.url) -}} {{- end -}} {{- $graph = $graph | append (dict "@type" "BreadcrumbList" "@id" $bcID "itemListElement" $items) -}} {{- $page = merge $page (dict "breadcrumb" (dict "@id" $bcID)) -}} {{- end -}} {{- /* Hauptentität */ -}} {{- $node := false -}} {{- if eq $main "Article" -}} {{- $artID := printf "%s#article" .Permalink -}} {{- $node = dict "@type" "Article" "@id" $artID "headline" (.Params.h1 | default .Title) "description" $desc "inLanguage" $lang "wordCount" .WordCount "mainEntityOfPage" (dict "@id" $pageID) "author" $authorRef "publisher" (dict "@id" $orgID) -}} {{- if not .Date.IsZero -}}{{- $node = merge $node (dict "datePublished" (.Date.Format "2006-01-02")) -}}{{- end -}} {{- if not .Lastmod.IsZero -}}{{- $node = merge $node (dict "dateModified" (.Lastmod.Format "2006-01-02")) -}}{{- end -}} {{- $page = merge $page (dict "mainEntity" (dict "@id" $artID)) -}} {{- else if eq $main "DefinedTermSet" -}} {{- $setID := printf "%s#glossary" .Permalink -}} {{- $terms := slice -}} {{- range .Pages.ByLinkTitle -}} {{- $terms = $terms | append (dict "@type" "DefinedTerm" "name" (.Params.term | default .Title) "url" .Permalink) -}} {{- end -}} {{- $node = dict "@type" "DefinedTermSet" "@id" $setID "name" (.Params.h1 | default .Title) "url" .Permalink "inLanguage" $lang -}} {{- if $terms -}}{{- $node = merge $node (dict "hasDefinedTerm" $terms) -}}{{- end -}} {{- $page = merge $page (dict "mainEntity" (dict "@id" $setID)) -}} {{- else if eq $main "DefinedTerm" -}} {{- $termID := printf "%s#term" .Permalink -}} {{- $node = dict "@type" "DefinedTerm" "@id" $termID "name" (.Params.term | default .Title) "url" .Permalink "inLanguage" $lang "author" $authorRef -}} {{- with (partial "helpers/first-paragraph.html" $) -}}{{- $node = merge $node (dict "description" .) -}}{{- end -}} {{- with site.GetPage "/glossar" -}}{{- $node = merge $node (dict "inDefinedTermSet" (dict "@id" (printf "%s#glossary" .Permalink))) -}}{{- end -}} {{- $page = merge $page (dict "mainEntity" (dict "@id" $termID)) -}} {{- end -}} {{- $graph = $graph | append $page -}} {{- if $node -}}{{- $graph = $graph | append $node -}}{{- end }} {{- end -}}
